% Compact chapter headings (also used by "Contents"), in the same style as section headings
\titleformat{\chapter}[block]{\py@HeaderFamily\huge\py@TitleColor}{\thechapter}{0.6em}{}[\udl@afterchapter]
\titlespacing*{\chapter}{0pt}{-24pt}{20pt}% negative: titlesec adds its own space above chapters

% Contents: chapter lines without a page number (report class \l@chapter minus the number)
\renewcommand*\l@chapter[2]{%
  \ifnum \c@tocdepth >\m@ne
    \addpenalty{-\@highpenalty}%
    \vskip 1.0em \@plus\p@
    \setlength\@tempdima{1.5em}%
    \begingroup
      \parindent \z@ \rightskip \@pnumwidth \parfillskip \z@ \@plus 1fil
      \leavevmode \bfseries
      \advance\leftskip\@tempdima \hskip -\leftskip
      #1\par
      \penalty\@highpenalty
    \endgroup
  \fi}

% Header: Upside Down Labs logo and product name, alternating sides like a book
% (odd pages: logo left, name right; even pages: name left, logo right)
% The logo has the website link under it; #1 = l or r, the side it is aligned to
\geometry{headheight=33pt}
\newcommand{\udllogo}[1]{%
  \begin{tabular}[b]{@{}#1@{}}
    \includegraphics[height=18pt]{logo_black_full.pdf}\\[-1pt]
    \href{https://www.upsidedownlabs.in}{\color{black}\sffamily\fontsize{6.5}{7}\selectfont www.upsidedownlabs.in}
  \end{tabular}}
\newcommand{\udlheader}{%
  \fancyhead[LO]{\udllogo{l}}%
  \fancyhead[RE]{\udllogo{r}}%
  \fancyhead[RO,LE]{{\py@HeaderFamily \@title\sphinxheadercomma\py@release}}%
  \renewcommand{\headrulewidth}{0.4pt}%
  \renewcommand{\footrulewidth}{0.4pt}}

% First page heading in the same style as section headings. In the contents it is listed
% as a section of chapter 1 (1.1, 1.2, ...), pointing to page 1; chapter 1's own sections
% are numbered after them.
\newcounter{udlfront}
\def\udl@frontentries{}
\newcommand{\udlheading}[1]{%
  \par\stepcounter{udlfront}%
  \pdfbookmark[0]{#1}{udlfront.\arabic{udlfront}}%
  \edef\udl@tmp{\noexpand\udl@frontentry{\arabic{udlfront}}{\thepage}{\unexpanded{#1}}}%
  \expandafter\g@addto@macro\expandafter\udl@frontentries\expandafter{\udl@tmp}%
  {\Large\py@HeaderFamily\py@TitleColor\hypertarget{udlfront.\arabic{udlfront}}{#1}\par}\vspace{3mm}}

% Contents line for one first page heading: {number}{page}{title}
\newcommand{\udl@frontentry}[3]{%
  \edef\udl@tmp{\noexpand\addtocontents{toc}{\noexpand\protect\noexpand\contentsline{section}%
    {\noexpand\protect\noexpand\numberline{\thechapter.#1}\unexpanded{#3}}{#2}{udlfront.#1}}}%
  \udl@tmp}

% After the first numbered chapter's title: add the first page headings to the contents
% (right after the chapter's own entry) and continue section numbering after them
\newcommand{\udl@afterchapter}{%
  \ifnum\c@chapter=\@ne \ifnum\c@udlfront>\z@
    \udl@frontentries
    \setcounter{section}{\value{udlfront}}%
    \global\let\udl@frontentries\@empty \setcounter{udlfront}{0}%
  \fi\fi}

% The summary sections of a product's docs page (marked with udlsummary-<pdf name> in its
% index.rst) are on that product's datasheet first page, so they are left out of its own PDF.
% Other PDFs that include the page (e.g. kits) keep them.
\usepackage{comment}
\newcommand{\udlsummary}[1]{%
  \ifnum\pdfstrcmp{\jobname}{#1}=0 \excludecomment{udlsummary-#1}\else\includecomment{udlsummary-#1}\fi}

% First page lists
\usepackage{multicol}
\newcommand{\udllist}[1]{\begin{itemize}\setlength{\itemsep}{3pt}#1\end{itemize}}
\newcommand{\udlfeatures}{\udlheading{Features}\udllist{\udlget{features}}}
\newcommand{\udlapplications}{\udlheading{Applications}\udllist{\udlget{applications}}}

% Use the datasheet first page when the PDF has a .yml, otherwise Sphinx's title page
\newcommand{\udlmaketitle}{%
  \udlifset{datasheet}{\udldatasheettitle}{\sphinxmaketitle}}

% With a datasheet first page, pages are numbered 1, 2, 3, ... throughout (first page,
% contents, chapters), so contents entries for the first page point to page 1.
% Otherwise keep Sphinx's roman-numbered contents.
\newcommand{\udltableofcontents}{%
  \udlifset{datasheet}{%
    \begingroup
      \parskip \z@skip
      \sphinxtableofcontentshook
      \tableofcontents
    \endgroup
    \clearpage}{\sphinxtableofcontents}}

\newcommand{\udldatasheettitle}{%
  \begingroup
    \thispagestyle{udlfirstpage}%
    \setlength{\parindent}{0pt}%
    \begingroup % for PDF information dictionary
      \def\endgraf{ }\def\and{\& }%
      \pdfstringdefDisableCommands{\def\\{, }}%
      \hypersetup{pdfauthor={\@author}, pdftitle={\@title}}%
    \endgroup
    % Title and tagline
    {\py@HeaderFamily\huge \@title\par}
    \udlifset{tagline}{\vspace{2mm}{\raggedright\sffamily\large \udlget{tagline}\par}}{}%
    \vspace{3mm}
    \rule{\textwidth}{0.4pt}\par
    \vspace{5mm}
    % Overview (left half), product image (right half); same split as features/applications
    \begin{minipage}[c]{0.48\textwidth}
      \udlifset{overview}{%
        \udlheading{Overview}
        \udlget{overview}\par
        \udlifset{note}{\vspace{2mm}\textbf{Note:} \udlget{note}\par}{}}{}%
    \end{minipage}\hfill
    \begin{minipage}[c]{0.48\textwidth}\centering
      \udlifset{image}{%
        \includegraphics[width=\linewidth,height=6cm,keepaspectratio]{\udlget{image}}}{}%
    \end{minipage}\par
    \vspace{7mm}
    % Features (left), applications (right) when both are given, otherwise full width
    \udlifset{features}{%
      \udlifset{applications}{%
        \begin{minipage}[t]{0.48\textwidth}\vspace{0pt}\raggedright\udlfeatures\end{minipage}\hfill
        \begin{minipage}[t]{0.48\textwidth}\vspace{0pt}\raggedright\udlapplications\end{minipage}\par
      }{\udlfeatures}%
      \vspace{7mm}}{%
      \udlifset{applications}{\udlapplications\vspace{7mm}}{}}%
    % Kit contents, in two columns
    \udlifset{contents}{%
      \udlheading{Kit Contents}
      \begin{multicols}{2}\raggedright\udllist{\udlget{contents}}\end{multicols}
      \vspace{3mm}}{}%
    % Specifications, in the same table style as the rest of the document
    \udlifset{specifications}{%
      \udlheading{Specifications}
      \begingroup
      \sphinxthistablewithglobalstyle
      \begin{tabular}{p{\dimexpr0.26\linewidth-2\tabcolsep\relax}p{\dimexpr0.74\linewidth-2\tabcolsep\relax}}
      \sphinxtoprule
      \sphinxtableatstartofbodyhook
      \udlget{specifications}
      \sphinxbottomrule
      \end{tabular}\par
      \endgroup}{}%
    \vfill
  \endgroup
  \clearpage
  \let\thanks\relax\let\maketitle\relax
}
